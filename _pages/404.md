{% comment %} One producer per mode. In direct-gtag mode the event is emitted by
   _includes/analytics-providers/google-gtag.html, where the real gtag() function
   exists: gtag.js ignores both plain objects and plain arrays pushed to
   dataLayer directly. Below is the GTM-mode producer, read by a Custom Event
   trigger. {% endcomment %}
{% if site.analytics.tag_manager.id and site.analytics.tag_manager.owns_analytics %}
<script>
  window.dataLayer = window.dataLayer || [];
  window.dataLayer.push({
    event: 'page_not_found',
    page_location: window.location.href,
    page_path: window.location.pathname
  });
</script>
{% endif %}
